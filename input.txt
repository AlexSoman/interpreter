var|width|100
var|height|100
var|max iter|30
var|i|0
def|determine|x|y
    var|iter|0
    while|iter|<|max iter
        var|xtemp|x^2-y^2+x0
        var|y|2*x*y+y0
        var|x|xtemp
        var|last|x^2+y^2
        if|last|>|4
            return|"no
        done
        var|iter|iter+1
    done
    return|"yes
done
while|i|<|height
    var|j|0
    var|output|"
    while|j|<|width
        var|y0|(i/height)*2.24-1.12
        var|x0|(j/width)*2.47-2
        call|determine|x0|y0
        var|result|last
        if|result|=|"yes
            call|append|output|"#
        done
        if|result|=|"no
            call|append|output|" |
        done
        var|output|last
        var|j|j+1
    done
    call|print|output
    var|i|i+1
done
